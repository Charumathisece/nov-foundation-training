CREATE TABLE app_role (role_id INT PRIMARY KEY AUTO_INCREMENT, role_name VARCHAR(50) NOT NULL UNIQUE, description VARCHAR(255));

CREATE TABLE employee (employee_id INT PRIMARY KEY AUTO_INCREMENT, user_id INT NOT NULL UNIQUE, employee_code VARCHAR(30) UNIQUE, employee_name VARCHAR(100) NOT NULL, phone VARCHAR(15), department_id INT NOT NULL, current_job_title_id INT NOT NULL, target_job_title_id INT, manager_id INT, hire_date DATE, employment_status VARCHAR(20) NOT NULL, employee_employee_id INT NOT NULL, FOREIGN KEY (employee_employee_id) REFERENCES employee(employee_id) ON DELETE NO ACTION ON UPDATE NO ACTION);

CREATE TABLE user_account (user_id INT PRIMARY KEY AUTO_INCREMENT, username VARCHAR(100) NOT NULL UNIQUE, email VARCHAR(100) NOT NULL UNIQUE, password_hash VARCHAR(255) NOT NULL, role_id INT NOT NULL, current_status VARCHAR(20) NOT NULL, created_at DATETIME NOT NULL, app_role_role_id INT NOT NULL, employee_employee_id INT NOT NULL, FOREIGN KEY (app_role_role_id) REFERENCES app_role(role_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (employee_employee_id) REFERENCES employee(employee_id) ON DELETE NO ACTION ON UPDATE NO ACTION);

CREATE TABLE deparment (deparment_id INT PRIMARY KEY AUTO_INCREMENT, deparment_name VARCHAR(100) NOT NULL UNIQUE, description VARCHAR(255));

CREATE TABLE job_title (job_title_id INT PRIMARY KEY AUTO_INCREMENT, job_title_name VARCHAR(100) NOT NULL UNIQUE, description VARCHAR(255), deparment_deparment_id INT NOT NULL, employee_employee_id INT NOT NULL, FOREIGN KEY (deparment_deparment_id) REFERENCES deparment(deparment_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (employee_employee_id) REFERENCES employee(employee_id) ON DELETE NO ACTION ON UPDATE NO ACTION);

CREATE TABLE skill (skill_id INT PRIMARY KEY AUTO_INCREMENT, skill_name VARCHAR(50) NOT NULL UNIQUE, description VARCHAR(255), category VARCHAR(100));

CREATE TABLE job_title_skill (job_title_skill_id INT PRIMARY KEY AUTO_INCREMENT, job_title_id INT NOT NULL, skill_id INT NOT NULL, required_level INT NOT NULL, skill_skill_id INT NOT NULL, job_title_job_title_id INT NOT NULL, FOREIGN KEY (skill_skill_id) REFERENCES skill(skill_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (job_title_job_title_id) REFERENCES job_title(job_title_id) ON DELETE NO ACTION ON UPDATE NO ACTION);

CREATE TABLE employee_skill (employee_skill_id INT PRIMARY KEY AUTO_INCREMENT, employee_id INT NOT NULL, skill_id INT NOT NULL, current_level INT NOT NULL, last_updated DATE, assessed_by_user_id INT, user_account_user_id INT NOT NULL, employee_employee_id INT NOT NULL, skill_skill_id INT NOT NULL, FOREIGN KEY (user_account_user_id) REFERENCES user_account(user_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (employee_employee_id) REFERENCES employee(employee_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (skill_skill_id) REFERENCES skill(skill_id) ON DELETE NO ACTION ON UPDATE NO ACTION);

CREATE TABLE course (course_id INT PRIMARY KEY AUTO_INCREMENT, course_name VARCHAR(100) NOT NULL, description VARCHAR(255), duration_hours INT, seat_limit INT NOT NULL, status VARCHAR(20) NOT NULL, created_by_user_id INT NOT NULL, created_at DATETIME NOT NULL, user_account_user_id INT NOT NULL, FOREIGN KEY (user_account_user_id) REFERENCES user_account(user_id) ON DELETE NO ACTION ON UPDATE NO ACTION);

CREATE TABLE course_skill (course_skill_id INT PRIMARY KEY AUTO_INCREMENT, course_id INT NOT NULL, skill_id INT NOT NULL, level_gain INT NOT NULL, course_course_id INT NOT NULL, employee_employee_id INT NOT NULL, FOREIGN KEY (course_course_id) REFERENCES course(course_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (employee_employee_id) REFERENCES employee(employee_id) ON DELETE NO ACTION ON UPDATE NO ACTION);

CREATE TABLE course_enrollment (enrollment_id INT PRIMARY KEY AUTO_INCREMENT, employee_id INT NOT NULL, course_id INT NOT NULL, assigned_by_user_id INT, status VARCHAR(20) NOT NULL, created_at DATETIME NOT NULL, enrolled_at DATETIME, completion_date DATE, completion_score INT, user_account_user_id INT NOT NULL, course_course_id INT NOT NULL, employee_employee_id INT NOT NULL, FOREIGN KEY (user_account_user_id) REFERENCES user_account(user_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (course_course_id) REFERENCES course(course_id) ON DELETE NO ACTION ON UPDATE NO ACTION, FOREIGN KEY (employee_employee_id) REFERENCES employee(employee_id) ON DELETE NO ACTION ON UPDATE NO ACTION);
